cmake_minimum_required(VERSION 3.20)
project(A350AutoTaxi VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(AUTOTAXI_BUILD_PLUGIN "Build the X-Plane plugin" ON)
set(XPLANE_SDK "" CACHE PATH "X-Plane SDK root containing CHeaders and Libraries")
if(NOT XPLANE_SDK AND EXISTS "D:/XPLANESDK/CHeaders/XPLM/XPLMPlugin.h")
    set(XPLANE_SDK "D:/XPLANESDK")
endif()
add_library(autotaxi_core STATIC src/Geo.cpp src/AptDatabase.cpp src/CenterlineNetwork.cpp src/DsfLoader.cpp src/RoutePlanner.cpp src/TaxiController.cpp src/Config.cpp src/Panel.cpp src/ConsolasFont.cpp src/PushbackPlanner.cpp vendor/bp-geometry/Segments.cpp)
target_include_directories(autotaxi_core PRIVATE vendor/bp-geometry)
if(WIN32)
    target_link_libraries(autotaxi_core PUBLIC gdi32)
endif()
target_include_directories(autotaxi_core PUBLIC src)
set_target_properties(autotaxi_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(autotaxi_core PRIVATE /W4 /utf-8)
else()
    target_compile_options(autotaxi_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(AUTOTAXI_BUILD_PLUGIN)
    if(NOT EXISTS "${XPLANE_SDK}/CHeaders/XPLM/XPLMPlugin.h")
        message(FATAL_ERROR "Set -DXPLANE_SDK=path/to/SDK in CLion CMake options")
    endif()
    add_library(A350AutoTaxi MODULE src/Plugin.cpp src/ControlAdapter.cpp src/TaxiUI.cpp src/BetterPushbackAdapter.cpp)
    target_link_libraries(A350AutoTaxi PRIVATE autotaxi_core)
    target_include_directories(A350AutoTaxi PRIVATE "${XPLANE_SDK}/CHeaders/XPLM" "${XPLANE_SDK}/CHeaders/Widgets")
    target_compile_definitions(A350AutoTaxi PRIVATE XPLM200=1 XPLM210=1 XPLM300=1 XPLM301=1)
    set_target_properties(A350AutoTaxi PROPERTIES PREFIX "" SUFFIX ".xpl")
    if(WIN32)
        if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
            message(FATAL_ERROR "X-Plane requires a 64-bit toolchain")
        endif()
        target_compile_definitions(A350AutoTaxi PRIVATE IBM=1 APL=0 LIN=0 NOMINMAX WIN32_LEAN_AND_MEAN)
        target_link_libraries(A350AutoTaxi PRIVATE "${XPLANE_SDK}/Libraries/Win/XPLM_64.lib" opengl32)
        set_target_properties(A350AutoTaxi PROPERTIES OUTPUT_NAME "win" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/A350AutoTaxi/64")
        if(MINGW)
            target_link_options(A350AutoTaxi PRIVATE -static-libgcc -static-libstdc++ -static)
        endif()
    elseif(APPLE)
        target_compile_definitions(A350AutoTaxi PRIVATE IBM=0 APL=1 LIN=0)
        target_link_options(A350AutoTaxi PRIVATE "-undefined" "dynamic_lookup")
        target_link_libraries(A350AutoTaxi PRIVATE "-framework OpenGL")
        set_target_properties(A350AutoTaxi PROPERTIES OUTPUT_NAME "mac" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/A350AutoTaxi/64")
    else()
        target_compile_definitions(A350AutoTaxi PRIVATE IBM=0 APL=0 LIN=1)
        target_link_libraries(A350AutoTaxi PRIVATE GL)
        set_target_properties(A350AutoTaxi PROPERTIES OUTPUT_NAME "lin" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/A350AutoTaxi/64")
    endif()
    foreach(CONFIG_TYPE ${CMAKE_CONFIGURATION_TYPES})
        string(TOUPPER "${CONFIG_TYPE}" CONFIG_UPPER)
        set_target_properties(A350AutoTaxi PROPERTIES "LIBRARY_OUTPUT_DIRECTORY_${CONFIG_UPPER}" "${CMAKE_BINARY_DIR}/A350AutoTaxi/64")
    endforeach()
    add_custom_command(TARGET A350AutoTaxi POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CMAKE_SOURCE_DIR}/A350AutoTaxi.ini" "${CMAKE_BINARY_DIR}/A350AutoTaxi/A350AutoTaxi.ini")
    add_custom_command(TARGET A350AutoTaxi POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CMAKE_SOURCE_DIR}/vendor/bp-geometry/COPYING" "${CMAKE_BINARY_DIR}/A350AutoTaxi/BetterPushback-CDDL.txt"
        COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/vendor/bp-geometry"
        "${CMAKE_BINARY_DIR}/A350AutoTaxi/licenses/bp-geometry")
    if(MSVC)
        target_compile_options(A350AutoTaxi PRIVATE /W4 /utf-8)
    else()
        target_compile_options(A350AutoTaxi PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
