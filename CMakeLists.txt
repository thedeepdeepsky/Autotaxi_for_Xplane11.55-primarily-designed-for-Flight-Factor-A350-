    set_target_properties(A350AutoTaxi PROPERTIES PREFIX "" SUFFIX ".xpl")
    if(WIN32)
        if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
            message(FATAL_ERROR "X-Plane requires a 64-bit toolchain")
        endif()
        target_compile_definitions(A350AutoTaxi PRIVATE IBM=1 APL=0 LIN=0 NOMINMAX WIN32_LEAN_AND_MEAN)
        target_link_libraries(A350AutoTaxi PRIVATE "${XPLANE_SDK}/Libraries/Win/XPLM_64.lib" opengl32)
        set_target_properties(A350AutoTaxi PROPERTIES OUTPUT_NAME "win" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/A350AutoTaxi/64")
        if(MINGW)
            target_link_options(A350AutoTaxi PRIVATE -static-libgcc -static-libstdc++ -static)
        endif()
    elseif(APPLE)
        target_compile_definitions(A350AutoTaxi PRIVATE IBM=0 APL=1 LIN=0)
        target_link_options(A350AutoTaxi PRIVATE "-undefined" "dynamic_lookup")
        target_link_libraries(A350AutoTaxi PRIVATE "-framework OpenGL")
        set_target_properties(A350AutoTaxi PROPERTIES OUTPUT_NAME "mac" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/A350AutoTaxi/64")
    else()
        target_compile_definitions(A350AutoTaxi PRIVATE IBM=0 APL=0 LIN=1)
        target_link_libraries(A350AutoTaxi PRIVATE GL)
        set_target_properties(A350AutoTaxi PROPERTIES OUTPUT_NAME "lin" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/A350AutoTaxi/64")
    endif()
    foreach(CONFIG_TYPE ${CMAKE_CONFIGURATION_TYPES})
        string(TOUPPER "${CONFIG_TYPE}" CONFIG_UPPER)
        set_target_properties(A350AutoTaxi PROPERTIES "LIBRARY_OUTPUT_DIRECTORY_${CONFIG_UPPER}" "${CMAKE_BINARY_DIR}/A350AutoTaxi/64")
    endforeach()
    add_custom_command(TARGET A350AutoTaxi POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CMAKE_SOURCE_DIR}/A350AutoTaxi.ini" "${CMAKE_BINARY_DIR}/A350AutoTaxi/A350AutoTaxi.ini")
    add_custom_command(TARGET A350AutoTaxi POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different

        "${CMAKE_BINARY_DIR}/A350AutoTaxi/licenses/bp-geometry")

        target_compile_options(A350AutoTaxi PRIVATE /W4 /utf-8)
    else()
        target_compile_options(A350AutoTaxi PRIVATE -Wall -Wextra -Wpedantic)
    endif()
